import requests

def get_vt_results(url, api_key):
    print(f"Checking {url} with key {api_key}")
    headers = {"x-apikey": api_key}
    import hashlib
    url_id = hashlib.sha256(url.encode()).hexdigest()
    response = requests.get(f"https://www.virustotal.com/api/v3/urls/{url_id}", headers=headers)
    response.raise_for_status()
    return response.json()

## 2. Contract Prompt
Context: Python 3.12, FastAPI, async httpx. Existing helper encode_vt_url_id(url: str) -> str in src/services/vt_client.py. Stateless URL security scanner with zero disk persistence.
Task: Write an async function fetch_url_report(url: str, api_key: str, client: httpx.AsyncClient | None = None) -> dict | None that calls the VirusTotal v3 API endpoint GET https://www.virustotal.com/api/v3/urls/{id}.
Contract:
  - Encodes URL using encode_vt_url_id(url) to get the base64 ID.
  - Sends x-apikey: api_key header; never logs or includes api_key in error messages or exceptions.
  - Timeout: 5.0 seconds strictly enforced.
  - 200 OK: Returns dictionary containing only the last_analysis_stats (e.g., {"malicious": int, "suspicious": int, "harmless": int, "undetected": int}).
  - 404 Not Found: URL has not been analyzed yet; returns None (must not raise an exception).
  - 429 Too Many Requests: Raises custom UpstreamRateLimitError("VirusTotal API quota exceeded").
  - 5xx / Timeout / Network Disconnect: Raises custom UpstreamServiceError("VirusTotal upstream service unavailable").
Constraints:
  - Use httpx, no third-party SDKs (no virustotal-python).
  - No disk caching or file writes.
  - Provide typed exceptions VTClientError, UpstreamRateLimitError, and UpstreamServiceError.
Then list every assumption you made that I did not state.

### Output from Contract Prompt
import httpx
from src.services.vt_client import encode_vt_url_id

class VTClientError(Exception):
    pass

class UpstreamRateLimitError(VTClientError):
    pass

class UpstreamServiceError(VTClientError):
    pass

async def fetch_url_report(
    url: str, api_key: str, client: httpx.AsyncClient | None = None
) -> dict | None:
    vt_id = encode_vt_url_id(url)
    endpoint = f"https://www.virustotal.com/api/v3/urls/{vt_id}"
    headers = {"x-apikey": api_key, "Accept": "application/json"}

    should_close_client = False
    if client is None:
        client = httpx.AsyncClient(timeout=5.0)
        should_close_client = True

    try:
        response = await client.get(endpoint, headers=headers)
        if response.status_code == 200:
            payload = response.json()
            return payload.get("data", {}).get("attributes", {}).get("last_analysis_stats", {})
        if response.status_code == 404:
            return None
        if response.status_code == 429:
            raise UpstreamRateLimitError("VirusTotal API quota exceeded.")
        if response.status_code >= 500:
            raise UpstreamServiceError(f"VirusTotal server error (HTTP {response.status_code}).")
        raise UpstreamServiceError(f"Unexpected upstream response (HTTP {response.status_code}).")
    except httpx.TimeoutException as exc:
        raise UpstreamServiceError("Upstream connection to VirusTotal timed out.") from exc
    except httpx.RequestError as exc:
        raise UpstreamServiceError("Failed to connect to VirusTotal upstream.") from exc
    finally:
        if should_close_client:
            await client.aclose()

## 3. Assumptions Listed by Model
1. Assumed response dictionary paths always nest under data.attributes.last_analysis_stats.
2. Assumed client lifecycle instantiates and closes a single-use httpx.AsyncClient when client is None.
3. Assumed custom exceptions inherit from a common base VTClientError.
4. Assumed transport errors (httpx.RequestError) map directly to UpstreamServiceError.
